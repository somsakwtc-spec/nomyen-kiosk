        function detectMotion() {
            if (!state.isCameraOn) return;
            const isScreensaver = dom.screensaver.classList.contains('active');
            const sourceVideo = isScreensaver ? dom.screensaverVideo : dom.video;
            if (!isScreensaver && !dom.modal.classList.contains('invisible')) { requestAnimationFrame(detectMotion); return; }
            const cvs = document.getElementById('motion-canvas');
            if (!cvs || !sourceVideo) { requestAnimationFrame(detectMotion); return; }
            const ctx = cvs.getContext('2d');
            cvs.width = 64; cvs.height = 48;
            try {
                if(sourceVideo.readyState >= 2) {
                    ctx.drawImage(sourceVideo, 0, 0, cvs.width, cvs.height);
                    const cur = ctx.getImageData(0, 0, cvs.width, cvs.height);
                    if (state.previousFrame) {
                        let diff = 0;
                        const d = cur.data; const p = state.previousFrame.data;
                        for (let i = 0; i < d.length; i += 4) { if (Math.abs(d[i] - p[i]) + Math.abs(d[i + 1] - p[i + 1]) + Math.abs(d[i + 2] - p[i + 2]) > 100) diff++; }
                        if (diff > (cvs.width * cvs.height * 0.04)) {
                            if (isScreensaver) { wakeUp(); triggerGreeting(); } 
                            else if (state.isAutoGreet) {
                                if(window.resetIdleTimer) window.resetIdleTimer();
                                if (Date.now() - state.lastGreetTime > 15000) { triggerGreeting(); state.lastGreetTime = Date.now(); }
                                const w = document.getElementById('camera-widget');
                                if(w) { w.style.borderColor = '#10B981'; setTimeout(() => w.style.borderColor = '', 200); }
                            }
                        }
                    }
                    state.previousFrame = cur;
                }
            } catch (e) {}
            requestAnimationFrame(detectMotion);
        }

        function base64ToArrayBuffer(base64) {
            const binaryString = window.atob(base64);
            const len = binaryString.length;
            const bytes = new Uint8Array(len);
            for (let i = 0; i < len; i++) bytes[i] = binaryString.charCodeAt(i);
            return bytes.buffer;
        }

        async function speak(text) {
            if (!state.isSoundOn || !text) return;
            stopAllSpeech(); setVisualizer(true);

            // 1. ลองใช้ Gemini TTS (เสียงธรรมชาติกว่า)
            if (apiKey && !state.ttsFailed) {
                try {
                    const r = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/${geminiTTSModel}:generateContent?key=${apiKey}`, {
                        method: 'POST', headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify({ contents: [{ parts: [{ text: text }] }], generationConfig: { responseModalities: ["AUDIO"], speechConfig: { voiceConfig: { prebuiltVoiceConfig: { voiceName: "Kore" } } } } })
                    });
                    
                    if (!r.ok) { throw new Error(`API Error: ${r.status} ${r.statusText}`); }
                    const d = await r.json();
                    if (d.error) { throw new Error(`Gemini Error: ${d.error.message}`); }

                    const a = d?.candidates?.[0]?.content?.parts?.[0]?.inlineData?.data;
                    if (a) {
                        const w = pcmToWav(new Int16Array(base64ToArrayBuffer(a)));
                        state.currentAudio = new Audio(URL.createObjectURL(w));
                        state.currentAudio.onended = () => setVisualizer(false);
                        state.currentAudio.play();
                        return;
                    }
                } catch (e) { 
                    console.warn("TTS API Error (Switching to offline mode):", e);
                    // ไม่ต้องจำค่า error ถาวร เผื่อเน็ตกลับมาใช้ได้
                    // state.ttsFailed = true; 
                }
            }

            // 2. ใช้ Web Speech API (Offline / Fallback) พร้อมระบบแก้บั๊กเสียงหาย
            if ('speechSynthesis' in window) {
                const u = new SpeechSynthesisUtterance(text);
                u.lang = 'th-TH'; 
                const thVoice = state.systemVoices.find(v => v.lang.includes('th')) || state.systemVoices[0];
                if (thVoice) u.voice = thVoice;
                
                // ปรับค่าให้เสียงสมูทขึ้น (ค่า 1.0 คือปกติ)
                u.rate = 1.0; 
                u.pitch = 1.0; 
                u.volume = 1.0;

                u.onend = () => {
                    if(state.speechInterval) clearInterval(state.speechInterval);
                    setVisualizer(false);
                };
                
                u.onerror = (e) => {
                    console.error("Speech error:", e);
                    if(state.speechInterval) clearInterval(state.speechInterval);
                    setVisualizer(false);
                };

                window.speechSynthesis.speak(u);

                // FIX: Chrome มีบั๊กหยุดพูดเองเมื่อประโยคยาวเกิน 15 วินาที
                // เราใช้ setInterval เพื่อกระตุ้น (Pause/Resume) ให้ทำงานต่อ
                if (text.length > 80) {
                    state.speechInterval = setInterval(() => {
                        if (!window.speechSynthesis.speaking) {
                            clearInterval(state.speechInterval);
                            return;
                        }
                        window.speechSynthesis.pause();
                        window.speechSynthesis.resume();
                    }, 14000);
                }
            } else { setVisualizer(false); }
        }
        
        function pcmToWav(pcmData, sampleRate = 24000) {
            const numSamples = pcmData.length;
            const buffer = new ArrayBuffer(44 + numSamples * 2);
            const view = new DataView(buffer);
            const writeString = (view, offset, string) => { for (let i = 0; i < string.length; i++) view.setUint8(offset + i, string.charCodeAt(i)); };
            writeString(view, 0, 'RIFF');
            view.setUint32(4, 36 + numSamples * 2, true);
            writeString(view, 8, 'WAVE');
            writeString(view, 12, 'fmt ');
            view.setUint32(16, 16, true);
            view.setUint16(20, 1, true);
            view.setUint16(22, 1, true);
            view.setUint32(24, sampleRate, true);
            view.setUint32(28, sampleRate * 2, true);
            view.setUint16(32, 2, true);
            view.setUint16(34, 16, true);
            writeString(view, 36, 'data');
            view.setUint32(40, numSamples * 2, true);
            let offset = 44;
            for (let i = 0; i < numSamples; i++, offset += 2) view.setInt16(offset, pcmData[i], true);
            return new Blob([view], { type: 'audio/wav' });
        }
